# Public-source snapshot: original/local fixtures are deliberately not supplied.
# Register a test only when every absolute source-side input is present. Build
# directory arguments are outputs generated by tests and need not exist yet.
function(mgo2mt_test_if_inputs_present)
  foreach(_mgo2mt_arg IN LISTS ARGV)
    if(IS_ABSOLUTE "${_mgo2mt_arg}")
      cmake_path(NORMAL_PATH _mgo2mt_arg OUTPUT_VARIABLE _mgo2mt_path)
      cmake_path(IS_PREFIX CMAKE_BINARY_DIR "${_mgo2mt_path}" NORMALIZE _mgo2mt_is_output)
      if(NOT _mgo2mt_is_output AND NOT EXISTS "${_mgo2mt_path}")
        message(STATUS "Skipping ${ARGV1}: local input is not provided (${_mgo2mt_arg})")
        return()
      endif()
    endif()
  endforeach()
  add_test(${ARGV})
endfunction()
